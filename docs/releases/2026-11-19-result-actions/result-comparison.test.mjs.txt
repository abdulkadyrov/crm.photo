import test from 'node:test';import assert from 'node:assert/strict';import vm from 'node:vm';import {readFileSync} from 'node:fs';
const source=readFileSync(new URL('../../js/core/legacy-app.js',import.meta.url),'utf8');const body=source.slice(source.indexOf('async function showFinalWorkComparison('),source.indexOf('async function showFinalWorkViewer('));
function setup(original=true){let config,back,restored=0,revoked=[];const c={document:{activeElement:{}},Blob,mediaById:()=>original?{blob:new Blob(['camera'])}:null,ensureFinalWorkLoaded:async()=>({id:'w',sourceMediaId:'m'}),finalWorkImageBlob:()=>new Blob(['result']),captureFinalWorkDialogReturn:()=>()=>restored++,URL:{createObjectURL:blob=>blob.size===6?'blob:camera':'blob:result',revokeObjectURL:url=>revoked.push(url)},uid:()=> 'comparison',notify:message=>message,mountDialogHistory:options=>{back=options.restore;return{close:back};},openScanPhotoViewer:options=>{config=options;options.mountHistory(()=>{});return'viewer';}};vm.runInNewContext(body,c);return{c,get:()=>({config,back,restored,revoked})};}
test('direct comparison uses camera source, keeps URLs live, and restores origin on Back',async()=>{const t=setup();assert.equal(await t.c.showFinalWorkComparison('w',{}),'viewer');let s=t.get();assert.equal(s.config.compare,true);assert.deepEqual([...s.config.items].map(i=>i.label),['Оригинал','Результат']);assert.equal(s.revoked.length,0);s.back();s=t.get();assert.equal(s.restored,1);assert.equal(s.revoked.length,2);});
test('missing camera original is reported without substituting generated picture',async()=>{const t=setup(false);assert.match(await t.c.showFinalWorkComparison('w',{}),/исходная фотография/);assert.equal(t.get().config,undefined);});
